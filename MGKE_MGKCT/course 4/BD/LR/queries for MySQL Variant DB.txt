create table Специальности
(
Код int not null primary key auto_increment,
Название varchar(50) not null);

create table Отделения
(
Код int not null primary key auto_increment,
Код_специальности int not null,
Название varchar(50) not null,
foreign key (Код_специальности) references Специальности(Код));

create table Группы
(
Код int not null primary key auto_increment,
Код_отделения int not null,
foreign key (Код_отделения) references Отделения(Код));

create table Слушатели
(
Код int not null primary key auto_increment,
Код_группы int not null,
Фамилия varchar(50) not null,
Имя varchar(50) not null,
Отчество varchar(50) not null,
foreign key (Код_группы) references Группы(Код));

create table Преподаватели
(
Код int not null primary key auto_increment,
Фамилия varchar(50) not null,
Имя varchar(50) not null,
Отчество varchar(50) not null,
Телефон varchar(15) not null,
Стаж_работы int not null);

create table Занятия
(
Код int not null primary key auto_increment,
Название varchar(50) not null,
Вид_занятия varchar(10) not null,
Оплата_за_час decimal(10, 2) not null);

create table Часы_занятий
(
Код int not null primary key auto_increment,
Код_преподавателя int not null,
Код_группы int not null,
Код_дисциплины int not null,
Количество_часов int not null,
foreign key (Код_преподавателя) references Преподаватели(Код),
foreign key (Код_группы) references Группы(Код),
foreign key (Код_дисциплины) references Дисциплины(Код));

Вывести список слушателей, одной специальности;
select Слушатели.Код_группы, Слушатели.Фамилия, Слушатели.Имя, Слушатели.Отчество, Специальности.Название
from Слушатели, Группы, Отделения, Специальности where
Слушатели.Код_группы=Группы.Код
and
Группы.Код_отделения=Отделения.Код
and
Отделения.Код_специальности=Специальности.Код
and
Специальности.Название='Java Developer';

Реализовать функцию, которая возвращает количество групп;
DELIMITER $
CREATE FUNCTION Group_Count()
RETURNS int
DETERMINISTIC
BEGIN
  Declare count int default 0;
  Select Count(Код) into count from Группы;
  RETURN count;
END
$

Реализовать процедуру, которая возвращает список преподавателей, имеющих количество часов больше 30;
DELIMITER $
CREATE PROCEDURE Hours_Count()
BEGIN
  Select Фамилия, Имя, Отчество, Sum(Количество_часов) from Преподаватели, Часы_занятий where Преподаватели.Код = Часы_занятий.Код_преподавателя group by Фамилия;
END
$

Реализовать процедуру, которая выводит информацию о совокупных часах преподавания за месяц;
DELIMITER $
CREATE PROCEDURE AllHours_Count()
BEGIN

END
$


Реализовать процедуру, которая выводит информацию об открытых специальностях и отделений;
DELIMITER $
CREATE PROCEDURE SpecAndOtd()
BEGIN
  Select Специальности.Название as Название_Специальности, Отделения.Название Название_Отделения from Специальности, Отделения where Специальности.Код = Отделения.Код_специальности;
END
$

Реализовать триггер, ограничивающий количество слушателей в отделении;
DELIMITER $
CREATE TRIGGER Count_Hearers_Border Before Insert ON Слушатели
FOR EACH ROW
BEGIN
	Declare count Integer;
	SELECT COUNT(*) INTO count FROM Слушатели;
	IF count > 20 THEN
		SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'В отделении не может быть больше 20-ти человек';
	END IF;
END
$

Реализовать триггер, добавляющий преподавателей, которые отработали больше остальных в “премию”;
DELIMITER $
CREATE TRIGGER Add_In_Premia After Update ON Преподаватели FOR EACH ROW
BEGIN
	Declare topPrepodId int;
	SELECT Код INTO topPrepodId FROM Преподаватели order by Стаж_работы Limit 1;
	Insert Into Премия (Код_преподавателя) Values (topPrepodId);
END
$
